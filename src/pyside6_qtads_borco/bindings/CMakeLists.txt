# Generates and builds the `_pyside6_qtads_borco` extension module: shiboken6
# turns bindings.h + bindings.xml into the wrapper sources listed in
# GENERATED_SOURCES below, which are then compiled and linked against QtAds
# (qtadvanceddocking-qt6, from extern/qtads) and Qt.

find_package(Python COMPONENTS Interpreter Development.Module REQUIRED)

# Shiboken6Tools isn't found by find_package() out of the box, so locate it
# in the active Python environment's site-packages.
execute_process(
    COMMAND ${Python_EXECUTABLE} -c "import sysconfig; print(sysconfig.get_paths()['purelib'])"
    OUTPUT_VARIABLE Python_SITELIB
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
list(APPEND CMAKE_PREFIX_PATH "${Python_SITELIB}/shiboken6_generator/lib/cmake")

find_package(Shiboken6Tools REQUIRED)

set(QTADS_LIBRARY qtadvanceddocking-qt6)
set(QTADS_SRC_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../../../extern/qtads/src)

# shiboken writes into a directory per component of the typesystem's dotted
# package name (bindings.xml), and one wrapper per bound class or namespace.
set(GENERATED_DIR ${CMAKE_CURRENT_BINARY_DIR}/pyside6_qtads_borco/_pyside6_qtads_borco)
set(GENERATED_SOURCES
    ${GENERATED_DIR}/ads_wrapper.cpp
    ${GENERATED_DIR}/ads_cautohidedockcontainer_wrapper.cpp
    ${GENERATED_DIR}/ads_cautohidesidebar_wrapper.cpp
    ${GENERATED_DIR}/ads_cautohidetab_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockareatabbar_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockareatitlebar_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockareawidget_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockcomponentsfactory_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockcontainerwidget_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockfocuscontroller_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockingstatereader_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockmanager_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockoverlay_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockoverlaycross_wrapper.cpp
    ${GENERATED_DIR}/ads_cdocksplitter_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockwidget_wrapper.cpp
    ${GENERATED_DIR}/ads_cdockwidgettab_wrapper.cpp
    ${GENERATED_DIR}/ads_celidinglabel_wrapper.cpp
    ${GENERATED_DIR}/ads_cfloatingdockcontainer_wrapper.cpp
    ${GENERATED_DIR}/ads_cfloatingdragpreview_wrapper.cpp
    ${GENERATED_DIR}/ads_ciconprovider_wrapper.cpp
    ${GENERATED_DIR}/ads_cpushbutton_wrapper.cpp
    ${GENERATED_DIR}/ads_cresizehandle_wrapper.cpp
    ${GENERATED_DIR}/ads_cspacerwidget_wrapper.cpp
    ${GENERATED_DIR}/ads_ctitlebarbutton_wrapper.cpp
    ${GENERATED_DIR}/ads_ifloatingwidget_wrapper.cpp
    ${GENERATED_DIR}/_pyside6_qtads_borco_module_wrapper.cpp
)

# FORCE_LIMITED_API restricts the compile to Python's stable ABI
# (-DPy_LIMITED_API=0x030a0000), matching the .abi3 suffix the shiboken6
# toolchain gives the extension, so one wheel serves every Python >= 3.11.
shiboken_generator_create_binding(
    EXTENSION_TARGET _pyside6_qtads_borco
    GENERATED_SOURCES ${GENERATED_SOURCES}
    HEADERS ${CMAKE_CURRENT_SOURCE_DIR}/bindings.h
    TYPESYSTEM_FILE ${CMAKE_CURRENT_SOURCE_DIR}/bindings.xml
    LIBRARY_TARGET ${QTADS_LIBRARY}
    QT_MODULES Core Gui Widgets
    FORCE_LIMITED_API
    SHIBOKEN_EXTRA_OPTIONS
        -I${QTADS_SRC_DIR}
        # ads_version.h, generated by QtAds' CMake
        -I$<TARGET_PROPERTY:${QTADS_LIBRARY},BINARY_DIR>
)

if(NOT WIN32)
    # GCC/Clang default to exporting every symbol, which leaks shiboken's
    # internal (non-limited-API) converter helpers into the extension's
    # dynamic symbol table and fails cibuildwheel's abi3audit step on Linux.
    # The extension only needs to export its PyInit_ entry point.
    set_target_properties(_pyside6_qtads_borco PROPERTIES
        CXX_VISIBILITY_PRESET hidden
        VISIBILITY_INLINES_HIDDEN ON
    )
endif()

# _pyside6_qtads_borco.{pyd,so} links against QtAds; install both into the
# package directory and point the loader at that directory. COMPONENT python
# is the only component scikit-build-core installs (pyproject.toml).
if(APPLE)
    set_target_properties(_pyside6_qtads_borco PROPERTIES INSTALL_RPATH "@loader_path")
    set_target_properties(${QTADS_LIBRARY} PROPERTIES INSTALL_RPATH "@loader_path")
elseif(UNIX)
    set_target_properties(_pyside6_qtads_borco PROPERTIES INSTALL_RPATH "$ORIGIN")
    set_target_properties(${QTADS_LIBRARY} PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()

install(TARGETS _pyside6_qtads_borco
    DESTINATION pyside6_qtads_borco
    COMPONENT python
)
# COMPONENT binds to the artifact kind it follows, so each kind needs its own:
# RUNTIME is the Windows DLL, LIBRARY the Linux/macOS shared object.
install(TARGETS ${QTADS_LIBRARY}
    RUNTIME DESTINATION pyside6_qtads_borco COMPONENT python
    LIBRARY DESTINATION pyside6_qtads_borco COMPONENT python
)
